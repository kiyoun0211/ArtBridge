```typescript
// src/lib/supabase/storage.ts
// Signed URLs for the private artwork-originals bucket.
// Used by Phase 2 (artist artwork view) and Phase 4 (post-purchase delivery).
import { createAdminClient } from './admin';

/**
 * Generate a time-limited signed URL for a file in the private artwork-originals bucket.
 * Default TTL: 1 hour. Caller is responsible for not logging the URL.
 */
export async function getArtworkSignedUrl(
  storagePath: string,
  expiresIn = 3600,
): Promise<string> {
  const supabase = createAdminClient();
  const { data, error } = await supabase.storage
    .from('artwork-originals')
    .createSignedUrl(storagePath, expiresIn);

  if (error) throw error;
  return data.signedUrl;
}
```

This file imports `admin.ts` (service role) — note the security boundary: callers MUST be server-side (Route Handlers, Server Components, Server Actions). Add JSDoc warning at top: `/** SERVER-ONLY. Imports the service-role client. Never call from a Client Component. */`.

<threat_model>
## Trust Boundaries (Plan 02 scope)

| Boundary | Description |
|----------|-------------|
| anon API client → Postgres | Untrusted; gated by RLS on every table |
| authenticated API client → Postgres | Trusted only for own rows; gated by `auth.uid()` checks |
| auth.users INSERT (Supabase Auth managed) → public.profiles INSERT | Crosses schema boundary via SECURITY DEFINER trigger |
| anon → Storage public URL on private bucket | Must return 4xx |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Elevation of Privilege | Anon client reading any private row | mitigate | RLS enabled on every public table at CREATE TIME (`ENABLE ROW LEVEL SECURITY` inline); deny-by-default applies (Plan 05 verifies via Vitest) — SEC-01 |
| T-02-02 | Elevation of Privilege | User self-promoting to 'artist' role | mitigate | Role lives in `profiles` (RLS-protected). `profiles_update_own` policy allows row update but the table-level `CHECK (role IN ('artist','buyer'))` does not prevent role flips. Mitigation Phase 1.5: tighten `profiles_update_own` `WITH CHECK` to forbid role mutation (`OLD.role = NEW.role`) — DEFERRED to Phase 2 since Phase 1 has no UI to update the profile. Documented as accepted residual for Phase 1 only. |
| T-02-03 | Spoofing | Tampered `raw_user_meta_data->>'role'` value | mitigate | `profiles.role CHECK` constraint rejects anything not in ('artist','buyer'); transaction fails loudly. Default fallback is 'buyer' (least privilege). |
| T-02-04 | Information Disclosure | Public URL to `artwork-originals` returns content | mitigate | Bucket created with `public=false`; no permissive SELECT policy for anon on `storage.objects` for that bucket; access only via service-role-signed URLs (Plan 05 E2E test verifies 4xx) — SEC-02 |
| T-02-05 | Information Disclosure | `bids.bidder_id` leaked to anon (sniping aid) | mitigate | `bids_select_amounts_only` allows SELECT but query patterns expose only amount; `bidder_id` retrieval requires the `bids_select_own` policy (own rows only). Phase 5 must validate column-level expectations in views/queries. |
| T-02-06 | Tampering | RLS recursion via `profiles` subquery in `artworks` policy | mitigate | Replaced with `private.get_user_role()` SECURITY DEFINER helper per Pitfall 4 |
| T-02-07 | Information Disclosure | `idempotency_keys` accessed by users | mitigate | RLS on, ZERO permissive policies → anon and authenticated both get 0 rows; admin client (service role) bypasses RLS for webhook handlers (Phase 4) |
| T-02-08 | Tampering | Webhook replay (SEC-03) | accept (in Phase 1) | Schema column landed; enforcement is Phase 4 webhook handler — disposition flagged for tracking |

**Residual risk note for T-02-02:** Add `WITH CHECK` clause `(OLD.role = NEW.role OR OLD.role IS NULL)` to `profiles_update_own` policy in Phase 2 when profile-edit UI lands. Capture in Phase 2 RESEARCH cross-reference.
</threat_model>
