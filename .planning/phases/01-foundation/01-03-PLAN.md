   ```typescript
   export const SignupSchema = z.object({
     email: z.email({ message: '올바른 이메일 형식을 입력해 주세요.' }),
     password: z.string().min(8, { message: '비밀번호는 8자 이상이어야 합니다.' }),
     role: z.enum(['artist', 'buyer'], { message: '계정 유형을 선택해 주세요.' }),
   });
   export type SignupInput = z.infer<typeof SignupSchema>;

   export const LoginSchema = z.object({
     email: z.email({ message: '올바른 이메일 형식을 입력해 주세요.' }),
     password: z.string().min(1),
   });
   export type LoginInput = z.infer<typeof LoginSchema>;
   ```
   (Note Zod v4 idiom: top-level `z.email()` instead of `z.string().email()`. If `z.email` is not exported in installed v4 minor, fall back to `z.string().email({ message })` — both compile under 4.4.x.)

2. `src/actions/auth.ts` ('use server' at top):
   ```typescript
   'use server'
   import { redirect } from 'next/navigation';
   import { createClient } from '@/lib/supabase/server';
   import { SignupSchema, LoginSchema } from '@/lib/validation/auth';

   export type AuthState = { error: string | null };

   export async function signupAction(_prev: AuthState, formData: FormData): Promise<AuthState> {
     const parsed = SignupSchema.safeParse({
       email: formData.get('email'),
       password: formData.get('password'),
       role: formData.get('role'),
     });
     if (!parsed.success) {
       return { error: parsed.error.issues[0]?.message ?? '입력값을 확인해 주세요.' };
     }
     const supabase = await createClient();
     const { data, error } = await supabase.auth.signUp({
       email: parsed.data.email,
       password: parsed.data.password,
       options: { data: { role: parsed.data.role } },
     });
     if (error || !data.user) {
       // Generic error to prevent email enumeration
       return { error: '오류가 발생했습니다. 잠시 후 다시 시도해 주세요.' };
     }
     // Profile row is created by on_auth_user_created trigger (PLAN-02 migration 4).
     // If email confirmation is REQUIRED by Supabase Auth settings, data.session is null and the user
     //   is shown the email-sent UI before redirect. For local dev (auto-confirm on), session is set immediately.
     if (!data.session) {
       redirect('/auth/confirm-sent');  // Page renders the "이메일을 확인해 주세요" UI from UI-SPEC
     }
     redirect(parsed.data.role === 'artist' ? '/artist' : '/buyer');
   }

   export async function loginAction(_prev: AuthState, formData: FormData): Promise<AuthState> {
     const parsed = LoginSchema.safeParse({
       email: formData.get('email'),
       password: formData.get('password'),
     });
     if (!parsed.success) {
       return { error: '이메일 또는 비밀번호가 올바르지 않습니다.' };  // generic
     }
     const supabase = await createClient();
     const { data, error } = await supabase.auth.signInWithPassword({
       email: parsed.data.email,
       password: parsed.data.password,
     });
     if (error || !data.user) {
       return { error: '이메일 또는 비밀번호가 올바르지 않습니다.' };
     }
     const { data: profile } = await supabase
       .from('profiles')
       .select('role')
       .eq('id', data.user.id)
       .single();
     redirect(profile?.role === 'artist' ? '/artist' : '/buyer');
   }

   export async function logoutAction(): Promise<void> {
     const supabase = await createClient();
     await supabase.auth.signOut();
     redirect('/');
   }
   ```

## Trust Boundaries (Plan 03 scope)

| Boundary | Description |
|----------|-------------|
| browser form → Server Action | FormData crosses boundary; Zod validates server-side |
| Server Action → Supabase Auth | Email/password sent over TLS to Supabase Auth |
| auth.users INSERT → profiles INSERT | SECURITY DEFINER trigger (PLAN-02) bridges schemas |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Information Disclosure | Email enumeration via signup error revealing "email already in use" | mitigate | Generic error string "오류가 발생했습니다. 잠시 후 다시 시도해 주세요." returned for ALL signup failures regardless of cause |
| T-03-02 | Information Disclosure | Email enumeration via login distinguishing "wrong password" vs "no such user" | mitigate | Single generic message "이메일 또는 비밀번호가 올바르지 않습니다." for both failure modes |
| T-03-03 | Spoofing | Brute-force password attack | mitigate (delegated) | Supabase Auth applies rate limits at the Auth API; no custom auth code |
| T-03-04 | Tampering | CSRF on logout | mitigate | Logout uses `<form action={logoutAction}>` (POST). Next.js 16 Server Actions verify Origin header; reject cross-origin POSTs by default |
| T-03-05 | Tampering | Tampering with hidden role field at signup | mitigate | Zod enum validates server-side; Supabase trigger CHECK constraint enforces role ∈ ('artist','buyer') as defense-in-depth |
| T-03-06 | Elevation of Privilege | Client-side validation bypass | mitigate | All validation re-runs server-side in Server Action via `SignupSchema.safeParse(formData)` |
| T-03-07 | Information Disclosure | Stack traces / Supabase error messages leaked to client | mitigate | Server Actions return generic Korean strings only; raw `error.message` is never returned to the browser |
| T-03-08 | Repudiation | No audit log of auth events | accept | Supabase Auth logs are sufficient for MVP; Sentry hooks deferred. Re-evaluate at Phase 6. |
</threat_model>
