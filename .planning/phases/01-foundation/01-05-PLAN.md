2. `tests/rls/anon-access.test.ts`:
   ```typescript
   import { describe, it, expect, beforeAll, afterAll } from 'vitest';
   import { createClient as createSbClient } from '@supabase/supabase-js';

   const SUPABASE_URL = process.env.NEXT_PUBLIC_SUPABASE_URL!;
   const ANON_KEY = process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!;
   const SERVICE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY!;

   const anon = createSbClient(SUPABASE_URL, ANON_KEY);
   const admin = createSbClient(SUPABASE_URL, SERVICE_KEY, { auth: { persistSession: false } });

   let seededUserId: string;
   let seededArtworkId: string;

   beforeAll(async () => {
     // Create a seed user via Auth admin (this fires on_auth_user_created trigger)
     const { data: { user } } = await admin.auth.admin.createUser({
       email: `rls-seed-${Date.now()}@example.com`,
       password: 'rls-test-password',
       email_confirm: true,
       user_metadata: { role: 'artist' },
     });
     seededUserId = user!.id;
     // Insert a draft artwork (anon must NOT see drafts)
     const { data: artwork } = await admin.from('artworks').insert({
       artist_id: seededUserId,
       title: 'RLS seed draft',
       width_cm: 30, height_cm: 40,
       sale_type: 'fixed', price: 100000,
       status: 'draft',
     }).select().single();
     seededArtworkId = artwork!.id;
     // Insert a sample idempotency key
     await admin.from('idempotency_keys').insert({ key: `rls-seed-${Date.now()}` });
   });

   afterAll(async () => {
     await admin.from('artworks').delete().eq('id', seededArtworkId);
     await admin.auth.admin.deleteUser(seededUserId);
   });

   describe('SEC-01: anon Supabase client cannot read private rows', () => {
     it('profiles → 0 rows for anon', async () => {
       const { data } = await anon.from('profiles').select('*');
       expect(data ?? []).toHaveLength(0);
     });

     it('artworks → anon does NOT see draft rows (status filter active)', async () => {
       const { data } = await anon.from('artworks').select('id,status').eq('id', seededArtworkId);
       expect(data ?? []).toHaveLength(0);
     });

     it('artworks → anon CAN see public statuses if any exist (sanity, no leak of draft)', async () => {
       const { data } = await anon.from('artworks').select('status');
       const statuses = new Set((data ?? []).map(r => r.status));
       expect(statuses.has('draft')).toBe(false);
       expect(statuses.has('cancelled')).toBe(false);
     });

     it('orders → 0 rows for anon', async () => {
       const { data } = await anon.from('orders').select('*');
       expect(data ?? []).toHaveLength(0);
     });

     it('ai_jobs → 0 rows for anon', async () => {
       const { data } = await anon.from('ai_jobs').select('*');
       expect(data ?? []).toHaveLength(0);
     });

     it('idempotency_keys → 0 rows for anon (admin-only)', async () => {
       const { data } = await anon.from('idempotency_keys').select('*');
       expect(data ?? []).toHaveLength(0);
     });

     it('watchlist → 0 rows for anon', async () => {
       const { data } = await anon.from('watchlist').select('*');
       expect(data ?? []).toHaveLength(0);
     });

     it('auctions → anon sees only active/closed (no pending/cancelled)', async () => {
       const { data } = await anon.from('auctions').select('status');
       const statuses = new Set((data ?? []).map(r => r.status));
       expect(statuses.has('pending')).toBe(false);
       expect(statuses.has('cancelled')).toBe(false);
     });
   });
   ```

3. `tests/e2e/auth-flow.spec.ts` — golden path (Success Criteria #1, #2):
   ```typescript
   import { test, expect } from '@playwright/test';
   import { createClient } from '@supabase/supabase-js';

   const admin = createClient(
     process.env.NEXT_PUBLIC_SUPABASE_URL!,
     process.env.SUPABASE_SERVICE_ROLE_KEY!,
     { auth: { persistSession: false } }
   );

   test('signup as artist → redirect to /artist → reload preserves session → logout', async ({ page }) => {
     const email = `e2e-artist-${Date.now()}@example.com`;
     const password = 'test-password-123';

     // 1. Signup
     await page.goto('/signup');
     await page.getByLabel('이메일').fill(email);
     await page.getByLabel('비밀번호').fill(password);
     await page.getByRole('button', { name: '작가' }).click();
     await page.getByRole('button', { name: '가입하기' }).click();

     // 2. Lands on /artist
     await expect(page).toHaveURL(/\/artist$/);
     await expect(page.getByRole('heading', { level: 1 })).toContainText('반갑습니다');

     // 3. Reload preserves session
     await page.reload();
     await expect(page).toHaveURL(/\/artist$/);

     // 4. Logout
     await page.getByRole('button', { name: '로그아웃' }).click();
     await expect(page).toHaveURL(/\/$/);

     // Cleanup
     const { data: { users } } = await admin.auth.admin.listUsers();
     const u = users.find(x => x.email === email);
     if (u) await admin.auth.admin.deleteUser(u.id);
   });
   ```

   import { test, expect } from '@playwright/test';
   import { createClient } from '@supabase/supabase-js';

   const admin = createClient(
     process.env.NEXT_PUBLIC_SUPABASE_URL!,
     process.env.SUPABASE_SERVICE_ROLE_KEY!,

   test('unauthenticated /artist → /login (not 403)', async ({ page }) => {

     const email = `e2e-buyer-${Date.now()}@example.com`;

     // Now navigate to /artist
     const response = await page.goto('/artist');
     await expect(page).toHaveURL(/\/buyer$/);
     expect(response?.status()).toBeLessThan(400);

     // Cleanup
     const { data: { users } } = await admin.auth.admin.listUsers();
     const u = users.find(x => x.email === email);
     if (u) await admin.auth.admin.deleteUser(u.id);
   });
